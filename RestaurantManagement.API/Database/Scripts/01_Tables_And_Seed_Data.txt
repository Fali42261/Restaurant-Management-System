AS

 FROM dbo.Products p LEFT JOIN dbo.Categories c ON c.CategoryId=p.CategoryId

GO
CREATE OR ALTER VIEW dbo.vw_DailyBillTotals
AS
 SELECT CONVERT(date,CreatedAt) AS SaleDate,COUNT_BIG(*) AS BillCount,
        SUM(Total) AS GrossSales,SUM(Discount) AS Discounts,SUM(Tax) AS Tax
 FROM dbo.Bills GROUP BY CONVERT(date,CreatedAt);
GO

CREATE OR ALTER PROCEDURE dbo.usp_Products_ListPaged
 @PageNumber int=1,@PageSize int=25,@CategoryId uniqueidentifier=NULL,@Search nvarchar(160)=NULL
AS
BEGIN
 SET NOCOUNT ON;
 SET @PageNumber=CASE WHEN @PageNumber<1 THEN 1 ELSE @PageNumber END;
 SET @PageSize=CASE WHEN @PageSize<1 THEN 25 WHEN @PageSize>101 THEN 101 ELSE @PageSize END;
 SELECT p.ProductId,p.CategoryId,c.Name AS CategoryName,p.Name,p.Description,p.Price,p.ImageUrl,p.IsAvailable
 FROM dbo.Products p LEFT JOIN dbo.Categories c ON c.CategoryId=p.CategoryId
 WHERE p.IsDeleted=0 AND (@CategoryId IS NULL OR p.CategoryId=@CategoryId)
   AND (@Search IS NULL OR p.Name LIKE @Search+N'%')
 ORDER BY p.Name,p.ProductId
 OFFSET (@PageNumber-1)*@PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;
GO
CREATE OR ALTER PROCEDURE dbo.usp_Orders_ListPaged
 @PageNumber int=1,@PageSize int=25,@Status nvarchar(30)=NULL,@FromUtc datetime2=NULL,@ToUtc datetime2=NULL
AS
BEGIN
 SET NOCOUNT ON;
 SET @PageNumber=CASE WHEN @PageNumber<1 THEN 1 ELSE @PageNumber END;
 SET @PageSize=CASE WHEN @PageSize<1 THEN 25 WHEN @PageSize>101 THEN 101 ELSE @PageSize END;
 SELECT o.OrderId,o.OrderNumber,o.CustomerName,o.OrderType,o.TableId,o.Subtotal,o.Discount,o.Tax,o.Total,o.PaymentMethod,o.Status,o.CreatedAt,
        COALESCE(items.ItemSummary,N'') AS ItemSummary,COALESCE(items.Quantity,0) AS Quantity
 FROM dbo.Orders o
 OUTER APPLY
 (
  SELECT STRING_AGG(CONVERT(nvarchar(max),CONCAT(p.Name,N' × ',oi.Quantity)),N', ')
           WITHIN GROUP (ORDER BY p.Name) AS ItemSummary,
         SUM(oi.Quantity) AS Quantity
  FROM dbo.OrderItems oi INNER JOIN dbo.Products p ON p.ProductId=oi.ProductId
  WHERE oi.OrderId=o.OrderId
 ) items
 WHERE o.IsDeleted=0 AND (@Status IS NULL OR o.Status=@Status)
   AND (@FromUtc IS NULL OR o.CreatedAt>=@FromUtc) AND (@ToUtc IS NULL OR o.CreatedAt<@ToUtc)
 ORDER BY o.CreatedAt DESC,o.OrderId
 OFFSET (@PageNumber-1)*@PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;
GO

END;

CREATE OR ALTER PROCEDURE dbo.usp_KitchenTickets_ListPaged
 @PageNumber int=1,@PageSize int=26,@Status nvarchar(30)=NULL
AS
BEGIN
 SET NOCOUNT ON;
 SET @PageNumber=CASE WHEN @PageNumber<1 THEN 1 ELSE @PageNumber END;
 SET @PageSize=CASE WHEN @PageSize<1 THEN 26 WHEN @PageSize>101 THEN 101 ELSE @PageSize END;

 ;WITH TicketRows AS
 (
  SELECT k.TicketId,k.OrderId,o.OrderNumber,o.OrderType,o.TableId,
         COALESCE(t.Name,o.OrderType) AS Location,
         k.Status,k.CreatedAt,k.UpdatedAt,
         STRING_AGG(CONVERT(nvarchar(max),CONCAT(p.Name,N' × ',oi.Quantity)),N', ')
           WITHIN GROUP (ORDER BY p.Name) AS Items
  FROM dbo.KitchenTickets k
  INNER JOIN dbo.Orders o ON o.OrderId=k.OrderId
  LEFT JOIN dbo.RestaurantTables t ON t.TableId=o.TableId
  LEFT JOIN dbo.OrderItems oi ON oi.OrderId=o.OrderId
  LEFT JOIN dbo.Products p ON p.ProductId=oi.ProductId
  WHERE o.IsDeleted=0 AND (@Status IS NULL OR k.Status=@Status)
  GROUP BY k.TicketId,k.OrderId,o.OrderNumber,o.OrderType,o.TableId,
           t.Name,k.Status,k.CreatedAt,k.UpdatedAt
 )
 SELECT TicketId,OrderId,OrderNumber,OrderType,TableId,Location,
        Status,CreatedAt,UpdatedAt,COALESCE(Items,N'') AS Items
 FROM TicketRows
 ORDER BY CreatedAt DESC,TicketId
 OFFSET (@PageNumber-1)*@PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;
GO

/* Restaurant-scoped dashboard KPIs. Row-level security applies through the
   RestaurantId session context set by SqlDataAccess for every API connection. */
CREATE OR ALTER PROCEDURE dbo.usp_Dashboard_Summary
AS
BEGIN
 SET NOCOUNT ON;
 DECLARE @Today date=CONVERT(date,SYSUTCDATETIME());

 SELECT
  (SELECT COUNT(1) FROM dbo.Orders
   WHERE CreatedAt>=@Today AND CreatedAt<DATEADD(day,1,@Today) AND IsDeleted=0) AS TodayOrders,
  (SELECT COALESCE(SUM(AmountPaid),0) FROM dbo.Bills
   WHERE CreatedAt>=@Today AND CreatedAt<DATEADD(day,1,@Today)) AS TodaySales,
  (SELECT COUNT(1) FROM dbo.Reservations
   WHERE ReservationDate=@Today AND IsDeleted=0) AS TodayReservations,
  (SELECT COUNT(1) FROM dbo.RestaurantTables
   WHERE Status='Available' AND IsDeleted=0) AS AvailableTables,
  (SELECT COUNT(1) FROM dbo.Orders
   WHERE Status NOT IN('Completed','Cancelled') AND IsDeleted=0) AS ActiveOrders,
  (SELECT COUNT(1) FROM dbo.RestaurantTables WHERE IsDeleted=0) AS TotalTables,
  (SELECT COUNT(1) FROM dbo.RestaurantTables
   WHERE Status='Occupied' AND IsDeleted=0) AS OccupiedTables,
  (SELECT COALESCE(SUM(GuestCount),0) FROM dbo.Reservations
   WHERE ReservationDate>=@Today AND Status IN('Pending','Confirmed') AND IsDeleted=0) AS UpcomingGuests;
END;
GO

/* Dashboard analytics: date-bounded sales trend and top products. */
CREATE OR ALTER PROCEDURE dbo.usp_Dashboard_Sales
 @FromUtc datetime2,
 @ToUtcExclusive datetime2
AS
BEGIN
 SET NOCOUNT ON;
 SELECT CONVERT(date,CreatedAt) AS SaleDate,
        COUNT(1) AS BillCount,
        COALESCE(SUM(AmountPaid),0) AS GrossSales,
        COALESCE(SUM(Discount),0) AS Discounts,
        COALESCE(SUM(Tax),0) AS Tax
 FROM dbo.Bills
 WHERE CreatedAt>=@FromUtc AND CreatedAt<@ToUtcExclusive
 GROUP BY CONVERT(date,CreatedAt)
 ORDER BY SaleDate;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Dashboard_TopProducts
 @FromUtc datetime2,
 @ToUtcExclusive datetime2
AS
BEGIN
 SET NOCOUNT ON;
 SELECT TOP(10) p.Name,
        SUM(oi.Quantity) AS QuantitySold,
        SUM(oi.Quantity*oi.UnitPrice) AS Sales
 FROM dbo.OrderItems oi
 INNER JOIN dbo.Products p ON p.ProductId=oi.ProductId
 INNER JOIN dbo.Orders o ON o.OrderId=oi.OrderId
 WHERE o.IsDeleted=0
   AND o.CreatedAt>=@FromUtc AND o.CreatedAt<@ToUtcExclusive
   AND o.Status<>'Cancelled' AND p.IsDeleted=0
 GROUP BY p.Name
 ORDER BY QuantitySold DESC,p.Name;
END;
GO


CREATE OR ALTER PROCEDURE dbo.usp_KitchenTicket_UpdateStatus
 @TicketId uniqueidentifier,
 @Status nvarchar(30)
AS
BEGIN
 SET NOCOUNT ON;
 DECLARE @CurrentStatus nvarchar(30);
 SELECT @CurrentStatus=Status FROM dbo.KitchenTickets WHERE TicketId=@TicketId;
 IF @CurrentStatus IS NULL
 BEGIN
  SELECT CAST(0 AS int) AS UpdatedRows,N'NotFound' AS Outcome;
  RETURN;
 END;

 IF NOT (
   (@CurrentStatus='New' AND @Status IN('Preparing','Cancelled')) OR
   (@CurrentStatus='Preparing' AND @Status IN('Ready','Cancelled')) OR
   (@CurrentStatus='Ready' AND @Status IN('Served','Cancelled'))
 )
 BEGIN
  SELECT CAST(0 AS int) AS UpdatedRows,N'InvalidTransition' AS Outcome;
  RETURN;
 END;

 UPDATE dbo.KitchenTickets
 SET Status=@Status,UpdatedAt=SYSUTCDATETIME()
 WHERE TicketId=@TicketId AND Status=@CurrentStatus;

 DECLARE @UpdatedRows int=@@ROWCOUNT;
 SELECT @UpdatedRows AS UpdatedRows,
        CASE WHEN @UpdatedRows=1 THEN N'Updated' ELSE N'Conflict' END AS Outcome;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Pos_Checkout
 @OrderId uniqueidentifier,
 @OrderNumber nvarchar(40),
 @InvoiceNumber nvarchar(40),
 @CustomerName nvarchar(150)=NULL,
 @OrderType nvarchar(30),
 @TableId uniqueidentifier=NULL,
 @Discount decimal(18,2)=0,
 @Tax decimal(18,2)=0,
 @PaymentMethod nvarchar(30),
 @CashReceived decimal(18,2)=0,
 @CreatedBy uniqueidentifier,
 @ItemsJson nvarchar(max)
AS
BEGIN
 SET NOCOUNT ON;
 SET XACT_ABORT ON;

 IF ISJSON(@ItemsJson)<>1 THROW 51001,'Order items must be valid JSON.',1;
 IF @Discount<0 OR @Tax<0 OR @CashReceived<0 THROW 51002,'Discount, tax and cash received cannot be negative.',1;
 IF @OrderType NOT IN('DineIn','Takeaway','Delivery') THROW 51003,'Invalid order type.',1;
 IF @PaymentMethod NOT IN('Cash','UPI','Card','Online','Unpaid') THROW 51004,'Invalid payment method.',1;
 IF @OrderType='DineIn' AND @TableId IS NULL THROW 51005,'A table is required for dine-in orders.',1;

 DECLARE @Lines TABLE(ProductId uniqueidentifier NOT NULL,Quantity int NOT NULL,Notes nvarchar(300) NULL);
 INSERT @Lines(ProductId,Quantity,Notes)
 SELECT ProductId,Quantity,Notes
 FROM OPENJSON(@ItemsJson)
 WITH(ProductId uniqueidentifier '$.ProductId',Quantity int '$.Quantity',Notes nvarchar(300) '$.Notes');

 IF NOT EXISTS(SELECT 1 FROM @Lines) OR (SELECT COUNT(1) FROM @Lines)>100
  THROW 51006,'An order must contain 1 to 100 item lines.',1;
 IF EXISTS(SELECT 1 FROM @Lines WHERE ProductId IS NULL OR Quantity<1 OR Quantity>1000)
  THROW 51007,'Each item must have a valid product and quantity from 1 to 1000.',1;
 IF EXISTS
 (
  SELECT 1 FROM @Lines l
  LEFT JOIN dbo.Products p ON p.ProductId=l.ProductId AND p.IsDeleted=0 AND p.IsAvailable=1
  WHERE p.ProductId IS NULL
 )
  THROW 51008,'One or more products are unavailable in this restaurant.',1;
 IF @TableId IS NOT NULL AND NOT EXISTS
 (
  SELECT 1 FROM dbo.RestaurantTables WHERE TableId=@TableId AND IsDeleted=0 AND Status<>'OutOfService'
 )
  THROW 51009,'The selected table is unavailable in this restaurant.',1;

 DECLARE @Subtotal decimal(18,2)=
 (
  SELECT SUM(p.Price*l.Quantity)
  FROM @Lines l INNER JOIN dbo.Products p ON p.ProductId=l.ProductId
 );
 IF @Subtotal IS NULL OR @Discount>@Subtotal
  THROW 51010,'Discount cannot exceed the order subtotal.',1;

 DECLARE @Total decimal(18,2)=@Subtotal-@Discount+@Tax;
 IF @Total<=0 THROW 51011,'The payable amount must be greater than zero.',1;

 DECLARE @AmountPaid decimal(18,2)=
  CASE WHEN @PaymentMethod='Cash' THEN CASE WHEN @CashReceived>@Total THEN @Total ELSE @CashReceived END
       WHEN @PaymentMethod IN('UPI','Card','Online') THEN @Total ELSE 0 END;
 DECLARE @BillStatus nvarchar(30)=CASE WHEN @AmountPaid>=@Total THEN 'Paid' ELSE 'Unpaid' END;
 DECLARE @Guest nvarchar(150)=COALESCE(NULLIF(LTRIM(RTRIM(@CustomerName)),''),N'Walk-in Guest');

 BEGIN TRANSACTION;
 BEGIN TRY
  INSERT dbo.Orders(OrderId,OrderNumber,CustomerName,OrderType,TableId,Subtotal,Discount,Tax,Total,
                    PaymentMethod,CashReceived,Status,CreatedBy,CreatedAt,IsDeleted)
  VALUES(@OrderId,@OrderNumber,@Guest,@OrderType,@TableId,@Subtotal,@Discount,@Tax,@Total,
         @PaymentMethod,@CashReceived,'Placed',@CreatedBy,SYSUTCDATETIME(),0);

  INSERT dbo.OrderItems(OrderItemId,OrderId,ProductId,Quantity,UnitPrice,Notes)
  SELECT NEWID(),@OrderId,l.ProductId,l.Quantity,p.Price,l.Notes
  FROM @Lines l INNER JOIN dbo.Products p ON p.ProductId=l.ProductId;

  INSERT dbo.KitchenTickets(TicketId,OrderId,Status,CreatedAt)
  VALUES(NEWID(),@OrderId,'New',SYSUTCDATETIME());

  INSERT dbo.Bills(BillId,InvoiceNumber,OrderId,CustomerName,OrderType,Subtotal,Discount,Tax,Total,
                   PaymentMethod,AmountPaid,Status,CreatedAt)
  VALUES(NEWID(),@InvoiceNumber,@OrderId,@Guest,@OrderType,@Subtotal,@Discount,@Tax,@Total,
         @PaymentMethod,@AmountPaid,@BillStatus,SYSUTCDATETIME());

  COMMIT TRANSACTION;
  SELECT @OrderId AS OrderId,@OrderNumber AS OrderNumber,@InvoiceNumber AS InvoiceNumber,
         @Subtotal AS Subtotal,@Discount AS Discount,@Tax AS Tax,@Total AS Total,
         @AmountPaid AS AmountPaid,@BillStatus AS Status;
 END TRY
 BEGIN CATCH
  IF @@TRANCOUNT>0 ROLLBACK TRANSACTION;
  THROW;
 END CATCH
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Auth_RegisterOwner
    @UserId UNIQUEIDENTIFIER,
    @FullName NVARCHAR(150),
    @Email NVARCHAR(256),
    @MobileNumber NVARCHAR(25),
    @PasswordHash NVARCHAR(500)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

    BEGIN TRY
        BEGIN TRANSACTION;

        IF EXISTS
        (
            SELECT 1
            FROM dbo.Users WITH(UPDLOCK, HOLDLOCK)
            WHERE IsDeleted=0
              AND (Email=@Email OR MobileNumber=@MobileNumber)
        )
        BEGIN
            ROLLBACK TRANSACTION;

        END;

        INSERT dbo.Users
        (
            UserId, FullName, Email, MobileNumber, PasswordHash,
            Role, IsActive, IsDeleted, CreatedAt
        )
        VALUES
        (
            @UserId, @FullName, @Email, @MobileNumber, @PasswordHash,
            N'Owner', 1, 0, SYSUTCDATETIME()
        );

        COMMIT TRANSACTION;
        SELECT CAST(1 AS BIT) AS Success, CAST(N'Created' AS NVARCHAR(30)) AS ErrorCode;
    END TRY
    BEGIN CATCH
        IF XACT_STATE()<>0 ROLLBACK TRANSACTION;

        IF ERROR_NUMBER() IN(2601,2627)
        BEGIN
            SELECT CAST(0 AS BIT) AS Success, CAST(N'Duplicate' AS NVARCHAR(30)) AS ErrorCode;
            RETURN;
        END;
        THROW;
    END CATCH;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;

CREATE OR ALTER PROCEDURE dbo.usp_Auth_StartPasswordReset
    @Email NVARCHAR(256),
    @CodeHash CHAR(64)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

    DECLARE @UserId UNIQUEIDENTIFIER;

    SELECT TOP(1) @UserId=UserId
    FROM dbo.Users
    WHERE Email=@Email AND IsActive=1 AND IsDeleted=0;

    IF @UserId IS NULL
    BEGIN
        SELECT CAST(0 AS BIT) AS Found, CAST(0 AS BIT) AS CanSend;
        RETURN;
    END;

    IF EXISTS
    (
        SELECT 1 FROM dbo.PasswordResetCodes
        WHERE UserId=@UserId
          AND CreatedAt>DATEADD(MINUTE,-1,SYSUTCDATETIME())
    )
    BEGIN
        SELECT CAST(1 AS BIT) AS Found, CAST(0 AS BIT) AS CanSend;
        RETURN;
    END;

    BEGIN TRANSACTION;

    DELETE dbo.PasswordResetCodes WHERE UserId=@UserId;

    INSERT dbo.PasswordResetCodes(UserId,CodeHash,ExpiresAt,CreatedAt,AttemptCount)
    VALUES(@UserId,@CodeHash,DATEADD(MINUTE,10,SYSUTCDATETIME()),SYSUTCDATETIME(),0);

    COMMIT TRANSACTION;

    SELECT CAST(1 AS BIT) AS Found, CAST(1 AS BIT) AS CanSend;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Auth_ResetPassword
    @Email NVARCHAR(256),
    @CodeHash CHAR(64),
    @PasswordHash NVARCHAR(500)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

    DECLARE @UserId UNIQUEIDENTIFIER;
    DECLARE @Consumed TABLE(UserId UNIQUEIDENTIFIER);

    BEGIN TRANSACTION;

    SELECT TOP(1) @UserId=UserId
    FROM dbo.Users WITH(UPDLOCK, HOLDLOCK)
    WHERE Email=@Email AND IsActive=1 AND IsDeleted=0;

    IF @UserId IS NULL
    BEGIN
        ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) AS Success;
        RETURN;
    END;

    DELETE dbo.PasswordResetCodes
    OUTPUT deleted.UserId INTO @Consumed(UserId)
    WHERE UserId=@UserId
      AND CodeHash=@CodeHash
      AND ExpiresAt>SYSUTCDATETIME()
      AND AttemptCount<5;

    IF NOT EXISTS(SELECT 1 FROM @Consumed)
    BEGIN
        UPDATE dbo.PasswordResetCodes
        SET AttemptCount=AttemptCount+1
        WHERE UserId=@UserId AND ExpiresAt>SYSUTCDATETIME();

        DELETE dbo.PasswordResetCodes
        WHERE UserId=@UserId AND AttemptCount>=5;

        COMMIT TRANSACTION;
        SELECT CAST(0 AS BIT) AS Success;
        RETURN;
    END;

    UPDATE dbo.Users
    SET PasswordHash=@PasswordHash, UpdatedAt=SYSUTCDATETIME()
    WHERE UserId=@UserId AND IsActive=1 AND IsDeleted=0;

    IF @@ROWCOUNT<>1
    BEGIN
        ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) AS Success;
        RETURN;
    END;

    UPDATE dbo.RefreshTokens
    SET RevokedAt=SYSUTCDATETIME()
    WHERE UserId=@UserId AND RevokedAt IS NULL;

    COMMIT TRANSACTION;
    SELECT CAST(1 AS BIT) AS Success;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO

/* -------------------------------------------------------------------------
   7. ATOMIC REFRESH TOKEN ROTATION
   ------------------------------------------------------------------------- */
CREATE OR ALTER PROCEDURE dbo.usp_Auth_RotateRefreshToken
    @OldTokenHash CHAR(64),
    @NewTokenHash CHAR(64),
    @ExpiresAt DATETIME2
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

    DECLARE @Session TABLE
    (
        TokenId UNIQUEIDENTIFIER,
        UserId UNIQUEIDENTIFIER,
        RestaurantId UNIQUEIDENTIFIER
    );

    DECLARE @User TABLE
    (
        FullName NVARCHAR(150),
        Email NVARCHAR(256),
        Role NVARCHAR(30)
    );

    DECLARE @UserId UNIQUEIDENTIFIER;
    DECLARE @RestaurantId UNIQUEIDENTIFIER;
    DECLARE @EffectiveRole NVARCHAR(30);
    DECLARE @RestaurantName NVARCHAR(160);

    BEGIN TRANSACTION;

    UPDATE dbo.RefreshTokens WITH(UPDLOCK, ROWLOCK)
    SET RevokedAt=SYSUTCDATETIME()
    OUTPUT deleted.TokenId, deleted.UserId, deleted.RestaurantId
        INTO @Session(TokenId,UserId,RestaurantId)
    WHERE TokenHash=@OldTokenHash
      AND RevokedAt IS NULL
      AND ExpiresAt>SYSUTCDATETIME();

    IF NOT EXISTS(SELECT 1 FROM @Session)
    BEGIN
        ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) AS Success;
        RETURN;
    END;

    SELECT TOP(1) @UserId=UserId, @RestaurantId=RestaurantId
    FROM @Session;

    INSERT @User(FullName,Email,Role)
    SELECT FullName,Email,Role
    FROM dbo.Users
    WHERE UserId=@UserId AND IsActive=1 AND IsDeleted=0;

    IF NOT EXISTS(SELECT 1 FROM @User)
    BEGIN
        ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) AS Success;
        RETURN;
    END;

    SELECT @EffectiveRole=Role FROM @User;

    IF @RestaurantId IS NOT NULL
    BEGIN
        SELECT TOP(1)
            @EffectiveRole=ur.Role,
            @RestaurantName=r.Name
        FROM dbo.UserRestaurants ur
        INNER JOIN dbo.Restaurants r ON r.RestaurantId=ur.RestaurantId
        WHERE ur.UserId=@UserId
          AND ur.RestaurantId=@RestaurantId
          AND ur.IsActive=1
          AND r.IsActive=1;

        IF @EffectiveRole IS NULL
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) AS Success;
            RETURN;
        END;
    END;

    INSERT dbo.RefreshTokens(TokenId,UserId,RestaurantId,TokenHash,ExpiresAt,CreatedAt)
    VALUES(NEWID(),@UserId,@RestaurantId,@NewTokenHash,@ExpiresAt,SYSUTCDATETIME());

    COMMIT TRANSACTION;

    SELECT
        CAST(1 AS BIT) AS Success,
        @UserId AS UserId,
        @RestaurantId AS RestaurantId,
        @RestaurantName AS RestaurantName,
        @EffectiveRole AS Role,
        u.FullName,
        u.Email
    FROM @User u;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Auth_Logout
    @UserId UNIQUEIDENTIFIER
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY

    UPDATE dbo.RefreshTokens
    SET RevokedAt=SYSUTCDATETIME()
    WHERE UserId=@UserId AND RevokedAt IS NULL;

    SELECT @@ROWCOUNT AS RevokedSessions;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO
