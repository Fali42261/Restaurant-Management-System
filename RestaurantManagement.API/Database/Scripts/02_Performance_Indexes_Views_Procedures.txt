CREATE OR ALTER VIEW dbo.vw_ActiveProducts
AS
 SELECT p.ProductId,p.CategoryId,c.Name AS CategoryName,p.Name,p.Description,p.Price,p.ImageUrl,p.IsAvailable
 FROM dbo.Products p LEFT JOIN dbo.Categories c ON c.CategoryId=p.CategoryId
 WHERE p.IsDeleted=0;
GO
CREATE OR ALTER VIEW dbo.vw_DailyBillTotals
AS
 SELECT CONVERT(date,CreatedAt) AS SaleDate,COUNT_BIG(*) AS BillCount,
        SUM(Total) AS GrossSales,SUM(Discount) AS Discounts,SUM(Tax) AS Tax
 FROM dbo.Bills GROUP BY CONVERT(date,CreatedAt);
GO

CREATE OR ALTER PROCEDURE dbo.usp_Products_ListPaged
 @PageNumber int=1,@PageSize int=25,@CategoryId uniqueidentifier=NULL,@Search nvarchar(160)=NULL
AS
BEGIN
 SET NOCOUNT ON;
 SET @PageNumber=CASE WHEN @PageNumber<1 THEN 1 ELSE @PageNumber END;
 SET @PageSize=CASE WHEN @PageSize<1 THEN 25 WHEN @PageSize>101 THEN 101 ELSE @PageSize END;
 SELECT ProductId,CategoryId,Name,Description,Price,ImageUrl,IsAvailable
 FROM dbo.Products
 WHERE IsDeleted=0 AND (@CategoryId IS NULL OR CategoryId=@CategoryId)
   AND (@Search IS NULL OR Name LIKE @Search+N'%')
 ORDER BY Name,ProductId
 OFFSET (@PageNumber-1)*@PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_KitchenTickets_ListPaged
 @PageNumber int=1,@PageSize int=26,@Status nvarchar(30)=NULL
AS
BEGIN
 SET NOCOUNT ON;
 SET @PageNumber=CASE WHEN @PageNumber<1 THEN 1 ELSE @PageNumber END;
 SET @PageSize=CASE WHEN @PageSize<1 THEN 26 WHEN @PageSize>101 THEN 101 ELSE @PageSize END;

 ;WITH TicketRows AS
 (
  SELECT k.TicketId,k.OrderId,o.OrderNumber,o.OrderType,o.TableId,
         COALESCE(t.Name,o.OrderType) AS Location,
         k.Status,k.CreatedAt,k.UpdatedAt,
         STRING_AGG(CONCAT(p.Name,N' × ',oi.Quantity),N', ')
           WITHIN GROUP (ORDER BY p.Name) AS Items
  FROM dbo.KitchenTickets k
  INNER JOIN dbo.Orders o ON o.OrderId=k.OrderId
  LEFT JOIN dbo.RestaurantTables t ON t.TableId=o.TableId
  LEFT JOIN dbo.OrderItems oi ON oi.OrderId=o.OrderId
  LEFT JOIN dbo.Products p ON p.ProductId=oi.ProductId
  WHERE o.IsDeleted=0 AND (@Status IS NULL OR k.Status=@Status)
  GROUP BY k.TicketId,k.OrderId,o.OrderNumber,o.OrderType,o.TableId,
           t.Name,k.Status,k.CreatedAt,k.UpdatedAt
 )
 SELECT TicketId,OrderId,OrderNumber,OrderType,TableId,Location,
        Status,CreatedAt,UpdatedAt,COALESCE(Items,N'') AS Items
 FROM TicketRows
 ORDER BY CreatedAt DESC,TicketId
 OFFSET (@PageNumber-1)*@PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;
GO

/* Restaurant-scoped dashboard KPIs. Row-level security applies through the
   RestaurantId session context set by SqlDataAccess for every API connection. */
CREATE OR ALTER PROCEDURE dbo.usp_Dashboard_Summary
AS
BEGIN
 SET NOCOUNT ON;
 DECLARE @Today date=CONVERT(date,SYSUTCDATETIME());

 SELECT
  (SELECT COUNT(1) FROM dbo.Orders
   WHERE CreatedAt>=@Today AND CreatedAt<DATEADD(day,1,@Today) AND IsDeleted=0) AS TodayOrders,
  (SELECT COALESCE(SUM(AmountPaid),0) FROM dbo.Bills
   WHERE CreatedAt>=@Today AND CreatedAt<DATEADD(day,1,@Today)) AS TodaySales,
  (SELECT COUNT(1) FROM dbo.Reservations
   WHERE ReservationDate=@Today AND IsDeleted=0) AS TodayReservations,
  (SELECT COUNT(1) FROM dbo.RestaurantTables
   WHERE Status='Available' AND IsDeleted=0) AS AvailableTables,
  (SELECT COUNT(1) FROM dbo.Orders
   WHERE Status NOT IN('Completed','Cancelled') AND IsDeleted=0) AS ActiveOrders,
  (SELECT COUNT(1) FROM dbo.RestaurantTables WHERE IsDeleted=0) AS TotalTables,
  (SELECT COUNT(1) FROM dbo.RestaurantTables
   WHERE Status='Occupied' AND IsDeleted=0) AS OccupiedTables,
  (SELECT COALESCE(SUM(GuestCount),0) FROM dbo.Reservations
   WHERE ReservationDate>=@Today AND Status IN('Pending','Confirmed') AND IsDeleted=0) AS UpcomingGuests;
END;
GO
PRINT 'swapDigit dashboard summary procedure created.';

/* Dashboard analytics: date-bounded sales trend and top products. */
CREATE OR ALTER PROCEDURE dbo.usp_Dashboard_Sales
 @FromUtc datetime2,
 @ToUtcExclusive datetime2
AS
BEGIN
 SET NOCOUNT ON;
 SELECT CONVERT(date,CreatedAt) AS SaleDate,
        COUNT(1) AS BillCount,
        COALESCE(SUM(AmountPaid),0) AS GrossSales,
        COALESCE(SUM(Discount),0) AS Discounts,
        COALESCE(SUM(Tax),0) AS Tax
 FROM dbo.Bills
 WHERE CreatedAt>=@FromUtc AND CreatedAt<@ToUtcExclusive
 GROUP BY CONVERT(date,CreatedAt)
 ORDER BY SaleDate;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Dashboard_TopProducts
 @FromUtc datetime2,
 @ToUtcExclusive datetime2
AS
BEGIN
 SET NOCOUNT ON;
 SELECT TOP(10) p.Name,
        SUM(oi.Quantity) AS QuantitySold,
        SUM(oi.Quantity*oi.UnitPrice) AS Sales
 FROM dbo.OrderItems oi
 INNER JOIN dbo.Products p ON p.ProductId=oi.ProductId
 INNER JOIN dbo.Orders o ON o.OrderId=oi.OrderId
 WHERE o.IsDeleted=0
   AND o.CreatedAt>=@FromUtc AND o.CreatedAt<@ToUtcExclusive
   AND o.Status<>'Cancelled' AND p.IsDeleted=0
 GROUP BY p.Name
 ORDER BY QuantitySold DESC,p.Name;
END;
GO


CREATE OR ALTER PROCEDURE dbo.usp_KitchenTicket_UpdateStatus
 @TicketId uniqueidentifier,
 @Status nvarchar(30)
AS
BEGIN
 SET NOCOUNT ON;
 DECLARE @CurrentStatus nvarchar(30);
 SELECT @CurrentStatus=Status FROM dbo.KitchenTickets WHERE TicketId=@TicketId;
 IF @CurrentStatus IS NULL
 BEGIN
  SELECT CAST(0 AS int) AS UpdatedRows,N'NotFound' AS Outcome;
  RETURN;
 END;

 IF NOT (
   (@CurrentStatus='New' AND @Status IN('Preparing','Cancelled')) OR
   (@CurrentStatus='Preparing' AND @Status IN('Ready','Cancelled')) OR
   (@CurrentStatus='Ready' AND @Status IN('Served','Cancelled'))
 )
 BEGIN
  SELECT CAST(0 AS int) AS UpdatedRows,N'InvalidTransition' AS Outcome;
  RETURN;
 END;

 UPDATE dbo.KitchenTickets
 SET Status=@Status,UpdatedAt=SYSUTCDATETIME()
 WHERE TicketId=@TicketId AND Status=@CurrentStatus;

 DECLARE @UpdatedRows int=@@ROWCOUNT;
 SELECT @UpdatedRows AS UpdatedRows,
        CASE WHEN @UpdatedRows=1 THEN N'Updated' ELSE N'Conflict' END AS Outcome;
END;
GO
