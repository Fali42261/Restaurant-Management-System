CREATE OR ALTER VIEW dbo.vw_ActiveProducts
AS
 SELECT p.ProductId,p.CategoryId,c.Name AS CategoryName,p.Name,p.Description,p.Price,p.ImageUrl,p.IsAvailable
 FROM dbo.Products p LEFT JOIN dbo.Categories c ON c.CategoryId=p.CategoryId
 WHERE p.IsDeleted=0;
GO
CREATE OR ALTER VIEW dbo.vw_DailyBillTotals
AS
 SELECT CONVERT(date,CreatedAt) AS SaleDate,COUNT_BIG(*) AS BillCount,
        SUM(Total) AS GrossSales,SUM(Discount) AS Discounts,SUM(Tax) AS Tax
 FROM dbo.Bills GROUP BY CONVERT(date,CreatedAt);
GO

CREATE OR ALTER PROCEDURE dbo.usp_Products_ListPaged
 @PageNumber int=1,@PageSize int=25,@CategoryId uniqueidentifier=NULL,@Search nvarchar(160)=NULL
AS
BEGIN
 SET NOCOUNT ON;
 SET @PageNumber=CASE WHEN @PageNumber<1 THEN 1 ELSE @PageNumber END;
 SET @PageSize=CASE WHEN @PageSize<1 THEN 25 WHEN @PageSize>101 THEN 101 ELSE @PageSize END;
 SELECT ProductId,CategoryId,Name,Description,Price,ImageUrl,IsAvailable
 FROM dbo.Products
 WHERE IsDeleted=0 AND (@CategoryId IS NULL OR CategoryId=@CategoryId)
   AND (@Search IS NULL OR Name LIKE @Search+N'%')
 ORDER BY Name,ProductId
 OFFSET (@PageNumber-1)*@PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;
GO
CREATE OR ALTER PROCEDURE dbo.usp_Orders_ListPaged
 @PageNumber int=1,@PageSize int=25,@Status nvarchar(30)=NULL,@FromUtc datetime2=NULL,@ToUtc datetime2=NULL
AS
BEGIN
 SET NOCOUNT ON;
 SET @PageNumber=CASE WHEN @PageNumber<1 THEN 1 ELSE @PageNumber END;
 SET @PageSize=CASE WHEN @PageSize<1 THEN 25 WHEN @PageSize>101 THEN 101 ELSE @PageSize END;
 SELECT OrderId,OrderNumber,CustomerName,OrderType,TableId,Subtotal,Discount,Tax,Total,PaymentMethod,Status,CreatedAt
 FROM dbo.Orders
 WHERE IsDeleted=0 AND (@Status IS NULL OR Status=@Status)
   AND (@FromUtc IS NULL OR CreatedAt>=@FromUtc) AND (@ToUtc IS NULL OR CreatedAt<@ToUtc)
 ORDER BY CreatedAt DESC,OrderId
 OFFSET (@PageNumber-1)*@PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;
GO
CREATE OR ALTER PROCEDURE dbo.usp_Bills_ListPaged
 @PageNumber int=1,@PageSize int=25,@FromUtc datetime2=NULL,@ToUtc datetime2=NULL
AS
BEGIN
 SET NOCOUNT ON;
 SET @PageNumber=CASE WHEN @PageNumber<1 THEN 1 ELSE @PageNumber END;
 SET @PageSize=CASE WHEN @PageSize<1 THEN 25 WHEN @PageSize>101 THEN 101 ELSE @PageSize END;
 SELECT BillId,InvoiceNumber,OrderId,CustomerName,OrderType,Subtotal,Discount,Tax,Total,PaymentMethod,AmountPaid,Status,CreatedAt
 FROM dbo.Bills

END;

/* Bounded page procedures for remaining high-volume collections.
   API passes pageSize+1 (max 101) to calculate hasMore without COUNT(*) scans. */
CREATE OR ALTER PROCEDURE dbo.usp_InventoryItems_ListPaged
 @PageNumber int=1,@PageSize int=26,@Search nvarchar(160)=NULL
AS
BEGIN
 SET NOCOUNT ON;
 SET @PageNumber=CASE WHEN @PageNumber<1 THEN 1 ELSE @PageNumber END;
 SET @PageSize=CASE WHEN @PageSize<1 THEN 26 WHEN @PageSize>101 THEN 101 ELSE @PageSize END;
 SELECT ItemId,Name,Unit,CurrentStock,ReorderLevel,CostPrice,IsActive
 FROM dbo.InventoryItems
 WHERE IsDeleted=0 AND (@Search IS NULL OR Name LIKE @Search+N'%')
 ORDER BY Name,ItemId OFFSET (@PageNumber-1)*@PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;
GO
CREATE OR ALTER PROCEDURE dbo.usp_StockMovements_ListPaged
 @PageNumber int=1,@PageSize int=26,@ItemId uniqueidentifier=NULL
AS
BEGIN
 SET NOCOUNT ON;
 SET @PageNumber=CASE WHEN @PageNumber<1 THEN 1 ELSE @PageNumber END;
 SET @PageSize=CASE WHEN @PageSize<1 THEN 26 WHEN @PageSize>101 THEN 101 ELSE @PageSize END;
 SELECT m.MovementId,m.ItemId,i.Name AS ItemName,m.MovementType,m.Quantity,m.Reference,m.Notes,m.CreatedAt
 FROM dbo.StockMovements m INNER JOIN dbo.InventoryItems i ON i.ItemId=m.ItemId
 WHERE (@ItemId IS NULL OR m.ItemId=@ItemId)
 ORDER BY m.CreatedAt DESC,m.MovementId OFFSET (@PageNumber-1)*@PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;
GO
