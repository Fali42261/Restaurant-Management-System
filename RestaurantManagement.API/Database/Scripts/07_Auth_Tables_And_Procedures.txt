/* -------------------------------------------------------------------------

IF OBJECT_ID(N'dbo.RefreshTokens', N'U') IS NULL
BEGIN
    CREATE TABLE dbo.RefreshTokens
    (
        TokenId UNIQUEIDENTIFIER NOT NULL CONSTRAINT PK_RefreshTokens PRIMARY KEY,
        UserId UNIQUEIDENTIFIER NOT NULL,
        RestaurantId UNIQUEIDENTIFIER NULL,
        TokenHash CHAR(64) NOT NULL,
        ExpiresAt DATETIME2(7) NOT NULL,
        CreatedAt DATETIME2(7) NOT NULL CONSTRAINT DF_RefreshTokens_CreatedAt DEFAULT(SYSUTCDATETIME()),
        RevokedAt DATETIME2(7) NULL,
        CONSTRAINT FK_RefreshTokens_Users FOREIGN KEY(UserId) REFERENCES dbo.Users(UserId)
    );
END;
GO

IF COL_LENGTH(N'dbo.RefreshTokens', N'RestaurantId') IS NULL
    ALTER TABLE dbo.RefreshTokens ADD RestaurantId UNIQUEIDENTIFIER NULL;
GO

IF COL_LENGTH(N'dbo.RefreshTokens', N'RevokedAt') IS NULL
    ALTER TABLE dbo.RefreshTokens ADD RevokedAt DATETIME2(7) NULL;
GO

IF OBJECT_ID(N'dbo.PasswordResetCodes', N'U') IS NULL
BEGIN
    CREATE TABLE dbo.PasswordResetCodes
    (
        UserId UNIQUEIDENTIFIER NOT NULL CONSTRAINT PK_PasswordResetCodes PRIMARY KEY,
        CodeHash CHAR(64) NOT NULL,
        ExpiresAt DATETIME2(7) NOT NULL,
        CreatedAt DATETIME2(7) NOT NULL CONSTRAINT DF_PasswordResetCodes_CreatedAt DEFAULT(SYSUTCDATETIME()),
        AttemptCount INT NOT NULL CONSTRAINT DF_PasswordResetCodes_AttemptCount DEFAULT(0),
        CONSTRAINT FK_PasswordResetCodes_Users FOREIGN KEY(UserId) REFERENCES dbo.Users(UserId)
    );
END;
GO

IF COL_LENGTH(N'dbo.PasswordResetCodes', N'AttemptCount') IS NULL
    ALTER TABLE dbo.PasswordResetCodes ADD AttemptCount INT NOT NULL
        CONSTRAINT DF_PasswordResetCodes_AttemptCount DEFAULT(0) WITH VALUES;
GO

/* Restaurant-scoped refresh tokens must belong to a valid user membership. */
IF OBJECT_ID(N'dbo.UserRestaurants', N'U') IS NOT NULL
   AND NOT EXISTS(SELECT 1 FROM sys.foreign_keys WHERE name=N'FK_RefreshTokens_User_Restaurant')
BEGIN
    ALTER TABLE dbo.RefreshTokens WITH CHECK
        ADD CONSTRAINT FK_RefreshTokens_User_Restaurant
        FOREIGN KEY(UserId, RestaurantId)
        REFERENCES dbo.UserRestaurants(UserId, RestaurantId);
END;
GO

IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID(N'dbo.RefreshTokens') AND name=N'UX_RefreshTokens_TokenHash')
    CREATE UNIQUE INDEX UX_RefreshTokens_TokenHash ON dbo.RefreshTokens(TokenHash);
GO

IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID(N'dbo.RefreshTokens') AND name=N'IX_RefreshTokens_User_Active')
    CREATE INDEX IX_RefreshTokens_User_Active
        ON dbo.RefreshTokens(UserId, RevokedAt, ExpiresAt)
        INCLUDE(RestaurantId);
GO

IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID(N'dbo.PasswordResetCodes') AND name=N'IX_PasswordResetCodes_Expiry')
    CREATE INDEX IX_PasswordResetCodes_Expiry
        ON dbo.PasswordResetCodes(ExpiresAt);
GO

/* -------------------------------------------------------------------------
   2. OWNER REGISTRATION
   ------------------------------------------------------------------------- */
CREATE OR ALTER PROCEDURE dbo.usp_Auth_RegisterOwner
    @UserId UNIQUEIDENTIFIER,
    @FullName NVARCHAR(150),
    @Email NVARCHAR(256),
    @MobileNumber NVARCHAR(25),
    @PasswordHash NVARCHAR(500)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

    BEGIN TRY
        BEGIN TRANSACTION;

        IF EXISTS
        (
            SELECT 1
            FROM dbo.Users WITH(UPDLOCK, HOLDLOCK)
            WHERE IsDeleted=0
              AND (Email=@Email OR MobileNumber=@MobileNumber)
        )
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) AS Success, CAST(N'Duplicate' AS NVARCHAR(30)) AS ErrorCode;
            RETURN;
        END;

        INSERT dbo.Users
        (
            UserId, FullName, Email, MobileNumber, PasswordHash,
            Role, IsActive, IsDeleted, CreatedAt
        )
        VALUES
        (
            @UserId, @FullName, @Email, @MobileNumber, @PasswordHash,
            N'Owner', 1, 0, SYSUTCDATETIME()
        );

        COMMIT TRANSACTION;
        SELECT CAST(1 AS BIT) AS Success, CAST(N'Created' AS NVARCHAR(30)) AS ErrorCode;
    END TRY
    BEGIN CATCH
        IF XACT_STATE()<>0 ROLLBACK TRANSACTION;

        IF ERROR_NUMBER() IN(2601,2627)
        BEGIN
            SELECT CAST(0 AS BIT) AS Success, CAST(N'Duplicate' AS NVARCHAR(30)) AS ErrorCode;
            RETURN;
        END;
        THROW;
    END CATCH;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Auth_GetUserRestaurants
    @UserId UNIQUEIDENTIFIER
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY

    SELECT r.RestaurantId, r.Name, ur.Role
    FROM dbo.UserRestaurants ur
    INNER JOIN dbo.Restaurants r ON r.RestaurantId=ur.RestaurantId
    INNER JOIN dbo.Users u ON u.UserId=ur.UserId
    WHERE ur.UserId=@UserId
      AND ur.IsActive=1
      AND r.IsActive=1
      AND u.IsActive=1
      AND u.IsDeleted=0
    ORDER BY r.Name;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO

/* -------------------------------------------------------------------------
   4. REFRESH TOKEN CREATE + PASSWORD RECORD
   ------------------------------------------------------------------------- */
CREATE OR ALTER PROCEDURE dbo.usp_Auth_CreateRefreshToken
    @UserId UNIQUEIDENTIFIER,
    @RestaurantId UNIQUEIDENTIFIER=NULL,
    @TokenHash CHAR(64),
    @ExpiresAt DATETIME2
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

    IF NOT EXISTS
    (
        SELECT 1 FROM dbo.Users
        WHERE UserId=@UserId AND IsActive=1 AND IsDeleted=0
    )
        THROW 51101, 'Inactive or missing user account.', 1;

    IF @RestaurantId IS NOT NULL
       AND NOT EXISTS
       (
           SELECT 1 FROM dbo.UserRestaurants ur
           INNER JOIN dbo.Restaurants r ON r.RestaurantId=ur.RestaurantId
           WHERE ur.UserId=@UserId AND ur.RestaurantId=@RestaurantId
             AND ur.IsActive=1 AND r.IsActive=1
       )
        THROW 51102, 'Restaurant membership is invalid or inactive.', 1;

    INSERT dbo.RefreshTokens(TokenId,UserId,RestaurantId,TokenHash,ExpiresAt,CreatedAt)
    VALUES(NEWID(),@UserId,@RestaurantId,@TokenHash,@ExpiresAt,SYSUTCDATETIME());
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO

/* -------------------------------------------------------------------------
   5. CHANGE PASSWORD (revokes all refresh sessions)
   ------------------------------------------------------------------------- */
CREATE OR ALTER PROCEDURE dbo.usp_Auth_ChangePassword
    @UserId UNIQUEIDENTIFIER,
    @PasswordHash NVARCHAR(500)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

    BEGIN TRANSACTION;

    UPDATE dbo.Users
    SET PasswordHash=@PasswordHash, UpdatedAt=SYSUTCDATETIME()
    WHERE UserId=@UserId AND IsActive=1 AND IsDeleted=0;

    IF @@ROWCOUNT<>1
    BEGIN
        ROLLBACK TRANSACTION;
        SELECT 0 AS RowsAffected;
        RETURN;
    END;

    UPDATE dbo.RefreshTokens
    SET RevokedAt=SYSUTCDATETIME()
    WHERE UserId=@UserId AND RevokedAt IS NULL;

    COMMIT TRANSACTION;
    SELECT 1 AS RowsAffected;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Auth_StartPasswordReset
    @Email NVARCHAR(256),
    @CodeHash CHAR(64)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

    DECLARE @UserId UNIQUEIDENTIFIER;

    SELECT TOP(1) @UserId=UserId
    FROM dbo.Users
    WHERE Email=@Email AND IsActive=1 AND IsDeleted=0;

    IF @UserId IS NULL
    BEGIN
        SELECT CAST(0 AS BIT) AS Found, CAST(0 AS BIT) AS CanSend;
        RETURN;
    END;

    IF EXISTS
    (
        SELECT 1 FROM dbo.PasswordResetCodes
        WHERE UserId=@UserId
          AND CreatedAt>DATEADD(MINUTE,-1,SYSUTCDATETIME())
    )
    BEGIN
        SELECT CAST(1 AS BIT) AS Found, CAST(0 AS BIT) AS CanSend;
        RETURN;
    END;

    BEGIN TRANSACTION;

    DELETE dbo.PasswordResetCodes WHERE UserId=@UserId;

    INSERT dbo.PasswordResetCodes(UserId,CodeHash,ExpiresAt,CreatedAt,AttemptCount)
    VALUES(@UserId,@CodeHash,DATEADD(MINUTE,10,SYSUTCDATETIME()),SYSUTCDATETIME(),0);

    COMMIT TRANSACTION;

    SELECT CAST(1 AS BIT) AS Found, CAST(1 AS BIT) AS CanSend;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Auth_ResetPassword
    @Email NVARCHAR(256),
    @CodeHash CHAR(64),
    @PasswordHash NVARCHAR(500)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

    DECLARE @UserId UNIQUEIDENTIFIER;
    DECLARE @Consumed TABLE(UserId UNIQUEIDENTIFIER);

    BEGIN TRANSACTION;

    SELECT TOP(1) @UserId=UserId
    FROM dbo.Users WITH(UPDLOCK, HOLDLOCK)
    WHERE Email=@Email AND IsActive=1 AND IsDeleted=0;

    IF @UserId IS NULL
    BEGIN
        ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) AS Success;
        RETURN;
    END;

    DELETE dbo.PasswordResetCodes
    OUTPUT deleted.UserId INTO @Consumed(UserId)
    WHERE UserId=@UserId
      AND CodeHash=@CodeHash
      AND ExpiresAt>SYSUTCDATETIME()
      AND AttemptCount<5;

    IF NOT EXISTS(SELECT 1 FROM @Consumed)
    BEGIN
        UPDATE dbo.PasswordResetCodes
        SET AttemptCount=AttemptCount+1
        WHERE UserId=@UserId AND ExpiresAt>SYSUTCDATETIME();

        DELETE dbo.PasswordResetCodes
        WHERE UserId=@UserId AND AttemptCount>=5;

        COMMIT TRANSACTION;
        SELECT CAST(0 AS BIT) AS Success;
        RETURN;
    END;

    UPDATE dbo.Users
    SET PasswordHash=@PasswordHash, UpdatedAt=SYSUTCDATETIME()
    WHERE UserId=@UserId AND IsActive=1 AND IsDeleted=0;

    IF @@ROWCOUNT<>1
    BEGIN
        ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) AS Success;
        RETURN;
    END;

    UPDATE dbo.RefreshTokens
    SET RevokedAt=SYSUTCDATETIME()
    WHERE UserId=@UserId AND RevokedAt IS NULL;

    COMMIT TRANSACTION;
    SELECT CAST(1 AS BIT) AS Success;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO

/* -------------------------------------------------------------------------
   7. ATOMIC REFRESH TOKEN ROTATION
   ------------------------------------------------------------------------- */
CREATE OR ALTER PROCEDURE dbo.usp_Auth_RotateRefreshToken
    @OldTokenHash CHAR(64),
    @NewTokenHash CHAR(64),
    @ExpiresAt DATETIME2
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY

    DECLARE @Session TABLE
    (
        TokenId UNIQUEIDENTIFIER,
        UserId UNIQUEIDENTIFIER,
        RestaurantId UNIQUEIDENTIFIER
    );

    DECLARE @User TABLE
    (
        FullName NVARCHAR(150),
        Email NVARCHAR(256),
        Role NVARCHAR(30)
    );

    DECLARE @UserId UNIQUEIDENTIFIER;
    DECLARE @RestaurantId UNIQUEIDENTIFIER;
    DECLARE @EffectiveRole NVARCHAR(30);
    DECLARE @RestaurantName NVARCHAR(160);

    BEGIN TRANSACTION;

    UPDATE dbo.RefreshTokens WITH(UPDLOCK, ROWLOCK)
    SET RevokedAt=SYSUTCDATETIME()
    OUTPUT deleted.TokenId, deleted.UserId, deleted.RestaurantId
        INTO @Session(TokenId,UserId,RestaurantId)
    WHERE TokenHash=@OldTokenHash
      AND RevokedAt IS NULL
      AND ExpiresAt>SYSUTCDATETIME();

    IF NOT EXISTS(SELECT 1 FROM @Session)
    BEGIN
        ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) AS Success;
        RETURN;
    END;

    SELECT TOP(1) @UserId=UserId, @RestaurantId=RestaurantId
    FROM @Session;

    INSERT @User(FullName,Email,Role)
    SELECT FullName,Email,Role
    FROM dbo.Users
    WHERE UserId=@UserId AND IsActive=1 AND IsDeleted=0;

    IF NOT EXISTS(SELECT 1 FROM @User)
    BEGIN
        ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) AS Success;
        RETURN;
    END;

    SELECT @EffectiveRole=Role FROM @User;

    IF @RestaurantId IS NOT NULL
    BEGIN
        SELECT TOP(1)
            @EffectiveRole=ur.Role,
            @RestaurantName=r.Name
        FROM dbo.UserRestaurants ur
        INNER JOIN dbo.Restaurants r ON r.RestaurantId=ur.RestaurantId
        WHERE ur.UserId=@UserId
          AND ur.RestaurantId=@RestaurantId
          AND ur.IsActive=1
          AND r.IsActive=1;

        IF @EffectiveRole IS NULL
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) AS Success;
            RETURN;
        END;
    END;

    INSERT dbo.RefreshTokens(TokenId,UserId,RestaurantId,TokenHash,ExpiresAt,CreatedAt)
    VALUES(NEWID(),@UserId,@RestaurantId,@NewTokenHash,@ExpiresAt,SYSUTCDATETIME());

    COMMIT TRANSACTION;

    SELECT
        CAST(1 AS BIT) AS Success,
        @UserId AS UserId,
        @RestaurantId AS RestaurantId,
        @RestaurantName AS RestaurantName,
        @EffectiveRole AS Role,
        u.FullName,
        u.Email
    FROM @User u;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO

CREATE OR ALTER PROCEDURE dbo.usp_Auth_Logout
    @UserId UNIQUEIDENTIFIER
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY

    UPDATE dbo.RefreshTokens
    SET RevokedAt=SYSUTCDATETIME()
    WHERE UserId=@UserId AND RevokedAt IS NULL;

    SELECT @@ROWCOUNT AS RevokedSessions;

        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;

/* -------------------------------------------------------------------------
   9. VERIFICATION
   ------------------------------------------------------------------------- */
SELECT
    t.name AS AuthTable,
    c.name AS ColumnName,
    TYPE_NAME(c.user_type_id) AS DataType,
    c.is_nullable AS IsNullable
FROM sys.tables t
INNER JOIN sys.columns c ON c.object_id=t.object_id
WHERE t.name IN(N'RefreshTokens',N'PasswordResetCodes')
ORDER BY t.name,c.column_id;

SELECT p.name AS AuthStoredProcedure
FROM sys.procedures p
WHERE p.name LIKE N'usp_Auth_%'
ORDER BY p.name;
