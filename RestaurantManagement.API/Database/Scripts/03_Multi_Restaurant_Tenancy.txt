/*
  swapDigit multi-restaurant tenant migration.
  Run AFTER 01_Tables_And_Seed_Data.txt and 02_Performance_Indexes_Views_Procedures.txt.
  Existing records are assigned to one Legacy Restaurant. Review that mapping before production.
*/
USE swapDigitDb;
GO
IF OBJECT_ID('dbo.Restaurants','U') IS NULL
BEGIN
 CREATE TABLE dbo.Restaurants(
  RestaurantId UNIQUEIDENTIFIER NOT NULL CONSTRAINT PK_Restaurants PRIMARY KEY,
  OwnerUserId UNIQUEIDENTIFIER NULL,
  Name NVARCHAR(160) NOT NULL,
  IsActive BIT NOT NULL CONSTRAINT DF_Restaurants_IsActive DEFAULT(1),
  CreatedAt DATETIME2 NOT NULL CONSTRAINT DF_Restaurants_CreatedAt DEFAULT(SYSUTCDATETIME()),
  UpdatedAt DATETIME2 NULL,
  CONSTRAINT FK_Restaurants_Owner FOREIGN KEY(OwnerUserId) REFERENCES dbo.Users(UserId)
 );
END
GO
IF OBJECT_ID('dbo.UserRestaurants','U') IS NULL
BEGIN
 CREATE TABLE dbo.UserRestaurants(
  UserId UNIQUEIDENTIFIER NOT NULL,
  RestaurantId UNIQUEIDENTIFIER NOT NULL,
  Role NVARCHAR(30) NOT NULL,
  IsActive BIT NOT NULL CONSTRAINT DF_UserRestaurants_IsActive DEFAULT(1),
  CreatedAt DATETIME2 NOT NULL CONSTRAINT DF_UserRestaurants_CreatedAt DEFAULT(SYSUTCDATETIME()),
  CONSTRAINT PK_UserRestaurants PRIMARY KEY(UserId,RestaurantId),
  CONSTRAINT FK_UserRestaurants_User FOREIGN KEY(UserId) REFERENCES dbo.Users(UserId),
  CONSTRAINT FK_UserRestaurants_Restaurant FOREIGN KEY(RestaurantId) REFERENCES dbo.Restaurants(RestaurantId)
 );
 CREATE INDEX IX_UserRestaurants_Restaurant ON dbo.UserRestaurants(RestaurantId,IsActive) INCLUDE(UserId,Role);
END
GO
/* A legacy container preserves existing data; new owners create their own restaurants. */
IF NOT EXISTS(SELECT 1 FROM dbo.Restaurants)
BEGIN
 DECLARE @LegacyOwner UNIQUEIDENTIFIER=(SELECT TOP(1) UserId FROM dbo.Users WHERE Role='Owner' AND IsDeleted=0 ORDER BY CreatedAt);
 INSERT dbo.Restaurants(RestaurantId,OwnerUserId,Name)
 VALUES(NEWID(),@LegacyOwner,N'Legacy Restaurant');
END
GO
INSERT dbo.UserRestaurants(UserId,RestaurantId,Role)
SELECT u.UserId,r.RestaurantId,u.Role
FROM dbo.Users u CROSS JOIN (SELECT TOP(1) RestaurantId FROM dbo.Restaurants ORDER BY CreatedAt) r
WHERE u.Role IN('Owner','Manager','Staff')
 AND NOT EXISTS(SELECT 1 FROM dbo.UserRestaurants ur WHERE ur.UserId=u.UserId AND ur.RestaurantId=r.RestaurantId);
GO
/* Add tenant keys to every restaurant-owned business table. */
DECLARE @Tables TABLE(TableName SYSNAME PRIMARY KEY);
INSERT @Tables VALUES
('Categories'),('Products'),('Floors'),('RestaurantTables'),('Orders'),('OrderItems'),
('KitchenTickets'),('Reservations'),('Bills'),('InventoryItems'),('StockMovements'),
('RestaurantSettings'),('Subscriptions'),('SupportTickets');
DECLARE @Table SYSNAME,@Sql NVARCHAR(MAX);
DECLARE c CURSOR LOCAL FAST_FORWARD FOR SELECT TableName FROM @Tables;
OPEN c; FETCH NEXT FROM c INTO @Table;
WHILE @@FETCH_STATUS=0
BEGIN
 IF COL_LENGTH('dbo.'+@Table,'RestaurantId') IS NULL
 BEGIN
  SET @Sql=N'ALTER TABLE dbo.'+QUOTENAME(@Table)+N' ADD RestaurantId UNIQUEIDENTIFIER NULL;';
  EXEC sys.sp_executesql @Sql;
 END;
 SET @Sql=N'UPDATE dbo.'+QUOTENAME(@Table)+N' SET RestaurantId=(SELECT TOP(1) RestaurantId FROM dbo.Restaurants ORDER BY CreatedAt) WHERE RestaurantId IS NULL;';
 EXEC sys.sp_executesql @Sql;
 IF NOT EXISTS(SELECT 1 FROM sys.default_constraints WHERE parent_object_id=OBJECT_ID('dbo.'+@Table) AND name='DF_'+@Table+'_RestaurantId')
 BEGIN
  SET @Sql=N'ALTER TABLE dbo.'+QUOTENAME(@Table)+N' ADD CONSTRAINT '+QUOTENAME('DF_'+@Table+'_RestaurantId')+
    N' DEFAULT(TRY_CONVERT(uniqueidentifier,SESSION_CONTEXT(N''RestaurantId''))) FOR RestaurantId;';
  EXEC sys.sp_executesql @Sql;
 END;
 SET @Sql=N'ALTER TABLE dbo.'+QUOTENAME(@Table)+N' ALTER COLUMN RestaurantId UNIQUEIDENTIFIER NOT NULL;';
 EXEC sys.sp_executesql @Sql;
 IF NOT EXISTS(SELECT 1 FROM sys.foreign_keys WHERE name='FK_'+@Table+'_Restaurants')
 BEGIN
  SET @Sql=N'ALTER TABLE dbo.'+QUOTENAME(@Table)+N' WITH CHECK ADD CONSTRAINT '+QUOTENAME('FK_'+@Table+'_Restaurants')+
   N' FOREIGN KEY(RestaurantId) REFERENCES dbo.Restaurants(RestaurantId);';
  EXEC sys.sp_executesql @Sql;
 END;
 IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID('dbo.'+@Table) AND name='IX_'+@Table+'_RestaurantId')
 BEGIN
  SET @Sql=N'CREATE INDEX '+QUOTENAME('IX_'+@Table+'_RestaurantId')+N' ON dbo.'+QUOTENAME(@Table)+N'(RestaurantId);';
  EXEC sys.sp_executesql @Sql;
 END;
 FETCH NEXT FROM c INTO @Table;
END
CLOSE c; DEALLOCATE c;
GO
/* Settings keys must be unique per restaurant, not globally. */
IF NOT EXISTS
(
 SELECT 1 FROM sys.key_constraints kc
 INNER JOIN sys.index_columns ic ON ic.object_id=kc.parent_object_id AND ic.index_id=kc.unique_index_id
 INNER JOIN sys.columns col ON col.object_id=ic.object_id AND col.column_id=ic.column_id
 WHERE kc.parent_object_id=OBJECT_ID('dbo.RestaurantSettings') AND kc.type='PK'
 GROUP BY kc.name HAVING COUNT(1)=2 AND SUM(CASE WHEN col.name IN('RestaurantId','SettingKey') THEN 1 ELSE 0 END)=2
)
BEGIN
 DECLARE @Pk SYSNAME;
 SELECT @Pk=kc.name FROM sys.key_constraints kc WHERE kc.parent_object_id=OBJECT_ID('dbo.RestaurantSettings') AND kc.type='PK';
 IF @Pk IS NOT NULL EXEC(N'ALTER TABLE dbo.RestaurantSettings DROP CONSTRAINT '+QUOTENAME(@Pk));
 ALTER TABLE dbo.RestaurantSettings ADD CONSTRAINT PK_RestaurantSettings PRIMARY KEY(RestaurantId,SettingKey);
END
GO
/* Subscriptions are restaurant-scoped. One restaurant can have only one current active subscription. */
/* Preserve only the newest active subscription per legacy restaurant before enforcing the rule. */
;WITH ranked AS
(
 SELECT SubscriptionId,ROW_NUMBER() OVER(PARTITION BY RestaurantId ORDER BY CreatedAt DESC,SubscriptionId DESC) AS rn
 FROM dbo.Subscriptions WHERE Status IN('Active','Trial')
)
UPDATE s SET Status='Expired' FROM dbo.Subscriptions s INNER JOIN ranked r ON r.SubscriptionId=s.SubscriptionId WHERE r.rn>1;
GO
IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID('dbo.Subscriptions') AND name='UX_Subscriptions_Restaurant_Active')
 CREATE UNIQUE INDEX UX_Subscriptions_Restaurant_Active ON dbo.Subscriptions(RestaurantId) WHERE Status IN('Active','Trial');
GO

/* Keep refresh sessions bound to the selected restaurant during token rotation. */
IF COL_LENGTH('dbo.RefreshTokens','RestaurantId') IS NULL
 ALTER TABLE dbo.RefreshTokens ADD RestaurantId UNIQUEIDENTIFIER NULL;
GO
IF NOT EXISTS(SELECT 1 FROM sys.foreign_keys WHERE name='FK_RefreshTokens_Restaurants')
 ALTER TABLE dbo.RefreshTokens ADD CONSTRAINT FK_RefreshTokens_Restaurants FOREIGN KEY(RestaurantId) REFERENCES dbo.Restaurants(RestaurantId);
GO

/* Subscription request metadata; activation remains a separate payment/admin action. */
IF COL_LENGTH('dbo.Subscriptions','BillingCycle') IS NULL ALTER TABLE dbo.Subscriptions ADD BillingCycle NVARCHAR(20) NULL;
IF COL_LENGTH('dbo.Subscriptions','Amount') IS NULL ALTER TABLE dbo.Subscriptions ADD Amount DECIMAL(18,2) NULL;
IF COL_LENGTH('dbo.Subscriptions','PaymentReference') IS NULL ALTER TABLE dbo.Subscriptions ADD PaymentReference NVARCHAR(120) NULL;
GO

/* Password reset OTP abuse protection. */
IF COL_LENGTH('dbo.PasswordResetCodes','AttemptCount') IS NULL
 ALTER TABLE dbo.PasswordResetCodes ADD AttemptCount INT NOT NULL CONSTRAINT DF_PasswordResetCodes_AttemptCount DEFAULT(0);
GO

/* Tenant-local natural-key uniqueness. */
IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID('dbo.Categories') AND name='UX_Categories_Restaurant_Name')
 CREATE UNIQUE INDEX UX_Categories_Restaurant_Name ON dbo.Categories(RestaurantId,Name) WHERE IsDeleted=0;
GO
IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID('dbo.Floors') AND name='UX_Floors_Restaurant_Name')
 CREATE UNIQUE INDEX UX_Floors_Restaurant_Name ON dbo.Floors(RestaurantId,Name) WHERE IsDeleted=0;
GO
IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID('dbo.RestaurantTables') AND name='UX_Tables_Restaurant_Floor_Name')
 CREATE UNIQUE INDEX UX_Tables_Restaurant_Floor_Name ON dbo.RestaurantTables(RestaurantId,FloorId,Name) WHERE IsDeleted=0;
GO
IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID('dbo.Products') AND name='UX_Products_Restaurant_Name')
 CREATE UNIQUE INDEX UX_Products_Restaurant_Name ON dbo.Products(RestaurantId,Name) WHERE IsDeleted=0;
GO
IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID('dbo.InventoryItems') AND name='UX_InventoryItems_Restaurant_Name')
 CREATE UNIQUE INDEX UX_InventoryItems_Restaurant_Name ON dbo.InventoryItems(RestaurantId,Name) WHERE IsDeleted=0;
GO

/* A staff/manager identity is assigned to one active restaurant only. Owners may own many. */
IF NOT EXISTS(SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID('dbo.UserRestaurants') AND name='UX_UserRestaurants_OneStaffRestaurant')
 CREATE UNIQUE INDEX UX_UserRestaurants_OneStaffRestaurant ON dbo.UserRestaurants(UserId) WHERE IsActive=1 AND Role IN('Manager','Staff');
GO
